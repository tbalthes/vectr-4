export function toAppRow(user_id: string, t: PlaidTransaction) {
  return {
    user_id,
    aggregator_transaction_id: t.transaction_id,
    account_id: t.account_id,
    name: t.merchant_name || t.name || null,
    amount: t.amount, // if you store cents, multiply or convert here
    iso_currency_code: t.iso_currency_code || null,
    date: t.date,
    raw: t,
    removed: false,
    updated_at: new Date().toISOString(),
  };
}

export async function upsertAddedAndModified(
  client: SupabaseClient,
  user_id: string,
  added: PlaidTransaction[],
  modified: PlaidTransaction[],
) {
  const repo = transactionsTable(client);
  const rows = [
    ...(added || []).map(t => toAppRow(user_id, t)),
    ...(modified || []).map(t => toAppRow(user_id, t)),
  ];
  if (rows.length === 0) return;

  const err = await repo.bulkUpsert(rows, 500); // onConflict: user_id,aggregator_transaction_id inside repo
  if (err) {
    logger.error({ event: 'sync.upsert_failed', count: rows.length, error: err.message }, 'Upsert failed');
  }
}

export async function markRemoved(
  client: SupabaseClient,
  user_id: string,
  removedIds: string[],
) {
  if (!removedIds || removedIds.length === 0) return;
  const repo = transactionsTable(client);
  const err = await repo.markRemovedByAggregatorIds(user_id, removedIds);
  if (err) {
    logger.error({ event: 'sync.remove_failed', count: removedIds.length, error: err.message }, 'Removal mark failed');
  }
}

import { logger } from '@/lib/logger';
import type { SupabaseClient } from '@supabase/supabase-js';
import { transactionsSync } from './client';
import { upsertAddedAndModified, markRemoved } from './transactions';
import type { PlaidSyncRequest, SyncSummary } from './types';

// Adjust to your DB helper locations
import { cursorsTable } from '@/lib/db/cursors';

type RunSyncParams = {
  client: SupabaseClient;
  access_token: string;
  user_id: string;
  item_id: string;
  start_cursor?: string | null;
  pageSize?: number;
};

export async function runTransactionsSync(params: RunSyncParams): Promise<SyncSummary> {
  const { client, access_token, user_id, item_id, start_cursor, pageSize = 100 } = params;
  const startedAt = Date.now();

  logger.info({ event: 'sync.start', item_id, user_id, has_cursor: !!start_cursor, count: pageSize }, 'Starting Plaid /transactions/sync');

  let nextCursor = start_cursor ?? null;
  let totalAdded = 0;
  let totalModified = 0;
  let totalRemoved = 0;

  // Loop until has_more=false
  // Also handle rare mutation errors by restarting with the last known cursor.
  // eslint-disable-next-line no-constant-condition
  while (true) {
    let page;
    try {
      page = await transactionsSync({ access_token, cursor: nextCursor, count: pageSize });
    } catch (e: any) {
      const msg = String(e?.message || e);
      logger.error({ event: 'sync.fetch_error', item_id, user_id, error: msg }, 'Error fetching sync page');
      throw e;
    }

    logger.info({
      event: 'sync.page',
      item_id,
      added: page.added?.length || 0,
      modified: page.modified?.length || 0,
      removed: page.removed?.length || 0,
      has_more: page.has_more,
    }, 'Sync page received');

    // Upsert added/modified
    await upsertAddedAndModified(client, user_id, page.added || [], page.modified || []);
    // Apply removals
    const removedIds = (page.removed || []).map(r => r.transaction_id);
    await markRemoved(client, user_id, removedIds);

    // Update totals and advance cursor
    totalAdded += page.added?.length || 0;
    totalModified += page.modified?.length || 0;
    totalRemoved += page.removed?.length || 0;
    nextCursor = page.next_cursor ?? nextCursor;

    if (!page.has_more) break;
  }

  // Persist final cursor
  if (item_id && nextCursor) {
    const cursorRepo = cursorsTable(client);
    const { error: cErr } = await cursorRepo.setCursor(item_id, nextCursor);
    if (cErr) {
      logger.error({ event: 'sync.cursor_update_failed', item_id, final_cursor: nextCursor, error: cErr.message }, 'Failed to persist cursor');
    } else {
      logger.info({ event: 'sync.cursor_updated', item_id, final_cursor: nextCursor }, 'Cursor updated');
    }
  }

  const duration_ms = Date.now() - startedAt;
  const summary: SyncSummary = {
    ok: true,
    item_id,
    user_id,
    added_count: totalAdded,
    modified_count: totalModified,
    removed_count: totalRemoved,
    final_cursor: nextCursor ?? null,
    duration_ms,
  };

  logger.info({ event: 'sync.completed', ...summary }, 'Plaid sync completed');
  return summary;
}
